'use strict';
const fs=require('node:fs'),path=require('node:path'),zlib=require('node:zlib'),assert=require('node:assert/strict'),crypto=require('node:crypto');
const E=require('../src/ecology/engine.js'),S=E.S;
const root=path.resolve(__dirname,'..');
function meter(){return {n:0,E:0,B:0,M:0,edible:0,lowE:0,lowM:0};}
function stocks(w){const out={};for(const o of w.organisms){const a=out[o.species]??=meter();a.n++;for(const k of ['E','B','M'])a[k]+=o[k];a.edible+=S.edibleB(o);a.lowE+=o.E<8;a.lowM+=o.M<.2;}return {species:out,soilM:w.tiles.reduce((n,t)=>n+t.soilM,0),resourceM:w.resources.reduce((n,a)=>n+a.M,0),carcassM:w.carcasses.reduce((n,a)=>n+a.M,0)};}
function record(out,id,key,value=1){const a=out[id]??={};a[key]=(a[key]||0)+value;}
function inspectBirths(w,c,out){const maps=S.maps(w);for(const o of w.organisms){const s=S.spec(o),p=c.traits.get(o.id);if(s.group==='main')continue;const a=out[o.species]??={};for(const [key,yes] of Object.entries({alive:true,juvenile:p.ageStage==='juvenile',phaseLow:p.phase<2,cooldown:o.birthReady>w.turn,damage:c.records.get(o.id).damage>0,E:o.E<(s.group==='prey'?12:8),M:s.group!=='prey'&&o.M<.2,B:s.group!=='prey'&&o.B<3,eligible:E.R.eligible(w,o,c)}))if(yes)a[key]=(a[key]||0)+1;
 let available=0,ruleSites=0;for(const cell of S.nearby(w,o.cell)){if(maps.food.has(cell)||!S.oneStep(w,o.cell,cell)||!S.validHabitat(w,s,cell))continue;available++;const count=S.nearby(w,cell).filter(q=>maps.food.get(q)?.species===s.id).length;if((s.group==='fungus'?[1,2,3]:[2,3]).includes(count))ruleSites++;}
 if(available)a.hasVacancy=(a.hasVacancy||0)+1;if(ruleSites)a.hasRuleSite=(a.hasRuleSite||0)+1;if(ruleSites&&E.R.eligible(w,o,c))a.eligibleWithSite=(a.eligibleWithSite||0)+1;
 }}
// Synchronous hooks observe existing outcomes only. Restore every method even on error.
function observedStep(w){const row={turn:w.turn+1,feeding:{},birthGates:{},losses:{},actions:{}};const restore=[];
 function hook(object,key,observe){const old=object[key];restore.push(()=>object[key]=old);object[key]=function(...args){return observe(old,...args)};}
 hook(E.N,'feed',(fn,w,c,transfer)=>{const before=new Map(w.organisms.map(o=>[o.id,{E:o.E,M:o.M,HP:o.HP}]));fn(w,c,transfer);for(const o of w.organisms){const b=before.get(o.id);for(const key of ['E','M','HP'])record(row.feeding,o.species,'gain'+key,o[key]-b[key]);record(row.feeding,o.species,'light',c.records.get(o.id).light);for(const event of c.events.get(o.id)||[])if(event.type==='feed'){record(row.feeding,o.species,'bites');record(row.feeding,o.species,'sourceE',event.amount);record(row.feeding,o.species,'sourceM',event.material);record(row.feeding,o.species,'source/'+(event.sourceSpecies||event.tag));}}});
 hook(E.N,'payActions',(fn,w,c)=>{for(const o of w.organisms)record(row.actions,o.species,'reserved',c.costs.get(o.id).action);return fn(w,c);});
 hook(E.N,'settle',(fn,w,c,fee)=>{for(const o of w.organisms){const cost=c.costs.get(o.id);record(row.losses,o.species,'baseDemand',cost.base);record(row.losses,o.species,'densityDemand',cost.density);if(o.E<cost.base+cost.density)record(row.losses,o.species,'cannotPayBase');if(o.infection)record(row.losses,o.species,'infected');}return fn(w,c,fee);});
 hook(E.R,'reproduce',(fn,w,c)=>{inspectBirths(w,c,row.birthGates);return fn(w,c);});
 try{const next=E.step(w);row.stocks=stocks(next);row.deaths={};for(const d of next.last.deathCauses)record(row.deaths,d.species,d.reason);row.counts=next.stats.counts;row.births=next.tallies.births;row.fingerprint=E.fingerprint(next);return {world:next,row};}finally{for(const undo of restore.reverse())undo();}
}
function foodOnly(w){w=structuredClone(w);w.organisms=w.organisms.filter(o=>S.spec(o).group!=='main');for(const o of w.organisms)o.infection=null;w.searchQ=[];w.pending=[];w.messages=[];w.contactPairs=[];w.initialMain=0;w.config.enabled={...w.config.enabled,main:false,P:false,Q:false};const t=S.totals(w);w.ledger={initialE:t.E,initialM:t.M,solar:0,sinks:0,Mloss:0};w.stats=E.summarize(w);return w;}
function run(id,treatment,out,turns=500){const base=require('../operations/evidence/feeding-regression-v4.1.1/report.json').results.find(c=>c.id===id);assert(base,'unknown fixed case');let w=E.create(base.config);const expected=treatment==='full'?JSON.parse(zlib.gunzipSync(fs.readFileSync(path.join(root,'operations/evidence/feeding-regression-v4.1.1',id+'.json.gz')))):null;if(treatment==='food-only')w=foodOnly(w);else assert.equal(treatment,'full');
 const rows=[{turn:0,stocks:stocks(w),counts:w.stats.counts,fingerprint:E.fingerprint(w)}];if(expected)assert.equal(rows[0].fingerprint,expected[0].fingerprint);const initialConfig=structuredClone(w.config);for(let t=1;t<=turns;t++){const r=observedStep(w);w=r.world;assert.deepEqual(E.invariants(w),[]);if(expected)assert.equal(r.row.fingerprint,expected[t].fingerprint,'observer changed biological outcome at '+t);rows.push(r.row);if(t%50===0)console.log(JSON.stringify({id,treatment,turn:t,counts:w.stats.counts}));}
 fs.mkdirSync(out,{recursive:true});fs.writeFileSync(path.join(out,id+'-'+treatment+'.json.gz'),zlib.gzipSync(JSON.stringify(rows)));const summary={id,treatment,derivedControl:treatment==='food-only'?'Remove main organisms and all initial P/Q after identical terrain/food generation; reset finite ledger to remaining stocks. Not a production preset or replay input.':null,config:initialConfig,checkedStates:rows.length,matchedStates:expected?rows.length:null,final:w.stats,tallies:w.tallies,totals:rows.slice(1).reduce((all,r)=>{for(const key of ['feeding','birthGates','losses','actions','deaths'])for(const [sp,values]of Object.entries(r[key]))for(const [k,v]of Object.entries(values))record(all[key]??={},sp,k,v);return all;},{}),snapshots:rows.filter(r=>r.turn%50===0).map(r=>({turn:r.turn,counts:r.counts,stocks:r.stocks}))};fs.writeFileSync(path.join(out,id+'-'+treatment+'.json'),JSON.stringify(summary,null,2));return summary;}
module.exports={observedStep,stocks,foodOnly};
if(require.main===module){const id=process.argv[2]||'lake',treatment=process.argv[3]||'full',out=path.resolve(process.argv[4]||'artifacts/foodweb-diagnosis-v4.1.1');run(id,treatment,out);const hashes={};for(const f of ['tools/diagnose-foodweb.cjs',...['engine','state','nutrition','reproduction','behavior','parasites','catalogue'].map(n=>'src/ecology/'+n+'.js')])hashes[f]=crypto.createHash('sha256').update(fs.readFileSync(path.join(root,f))).digest('hex');fs.writeFileSync(path.join(out,'source-hashes.json'),JSON.stringify(hashes,null,2));}
